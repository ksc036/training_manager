Reuse the existing model registry, manifest-backed dataset selection, and checkpoint format, while generating preview images on demand instead of storing persistent test artifacts.

```python
# app/services/test_inference.py
from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path

from app.services.runs import RunService


@dataclass(frozen=True)
class TestableRunSummary:
    run_name: str
    run_dir: str
    model_name: str
    encoder_name: str
    dataset_version_id: int


class TestInferenceService:
    def __init__(self, project_root: Path) -> None:
        self.project_root = project_root
        self._run_service = RunService(project_root)

    def list_testable_runs(self) -> list[TestableRunSummary]:
        runs: list[TestableRunSummary] = []
        for run in self._run_service.list_runs():
            checkpoint_path = Path(run.run_dir) / "checkpoints" / "best.ckpt"
            if run.status != "completed":
                continue
            if run.trainer_backend != "external-script":
                continue
            if not checkpoint_path.is_file():
                continue
            runs.append(
                TestableRunSummary(
                    run_name=run.run_name,
                    run_dir=run.run_dir,
                    model_name=run.model_name,
                    encoder_name=run.encoder_name,
                    dataset_version_id=run.dataset_version_id,
                )
            )
        return runs
```

```python
@dataclass(frozen=True)
class TestSampleSummary:
    sample_id: str
    image_path: str
    mask_path: str


@dataclass(frozen=True)
class InferencePreviewResult:
    run_name: str
    sample_id: str
    original_data_url: str
    ground_truth_data_url: str
    prediction_data_url: str
    overlay_data_url: str


def _to_data_url(image: Image.Image) -> str:
    buffer = io.BytesIO()
    image.save(buffer, format="PNG")
    encoded = base64.b64encode(buffer.getvalue()).decode("ascii")
    return f"data:image/png;base64,{encoded}"
```

```python
# app/services/test_inference.py
def run_inference(self, run_name: str, sample_id: str) -> InferencePreviewResult:
    run = self._require_testable_run(run_name)
    sample = self._require_sample(run.dataset_version_id, sample_id)
    checkpoint = torch.load(
        Path(run.run_dir) / "checkpoints" / "best.ckpt",
        map_location="cpu",
        weights_only=False,
    )
    model = build_segmentation_model(run.model_name)
    model.load_state_dict(checkpoint["model_state_dict"], strict=False)
    model.eval()

    original = Image.open(sample.image_path).convert("RGB")
    gt = Image.open(sample.mask_path).convert("L")
    resized = original.resize((64, 64))
    array = np.asarray(resized, dtype=np.float32) / 255.0
    tensor = torch.from_numpy(array.transpose(2, 0, 1)).unsqueeze(0)

    with torch.no_grad():
        logits = model(tensor)
        prediction = (torch.sigmoid(logits) > 0.5).float()[0, 0].cpu().numpy()

    prediction_image = Image.fromarray((prediction * 255).astype("uint8"), mode="L").resize(original.size)
    gt_display = gt.resize(original.size)
    overlay = original.copy()
    overlay_pixels = overlay.load()
    pred_pixels = prediction_image.load()
    for y in range(original.size[1]):
        for x in range(original.size[0]):
            if pred_pixels[x, y] > 127:
                r, g, b = overlay_pixels[x, y]
                overlay_pixels[x, y] = (255, g // 2, b // 2)

    return InferencePreviewResult(
        run_name=run.run_name,
        sample_id=sample.sample_id,
        original_data_url=_to_data_url(original),
        ground_truth_data_url=_to_data_url(gt_display.convert("RGB")),
        prediction_data_url=_to_data_url(prediction_image.convert("RGB")),
        overlay_data_url=_to_data_url(overlay),
    )
```

```python
# app/web/routes_test.py
@router.post("/test", response_class=HTMLResponse)
async def run_test_inference(request: Request) -> HTMLResponse:
    form = await request.form()
    run_name = str(form.get("run_name", "")).strip()
    sample_id = str(form.get("sample_id", "")).strip()
    service = TestInferenceService(project_root=request.app.state.project_root)
    runs = service.list_testable_runs()
    selected_run = next((run for run in runs if run.run_name == run_name), None)
    samples = service.list_samples_for_run(run_name) if selected_run is not None else []

    if selected_run is None:
        return templates.TemplateResponse(
            request=request,
            name="test.html",
            context={"page_title": "Test", "runs": runs, "samples": [], "error_message": "Choose a valid run."},
            status_code=400,
        )

    try:
        result = service.run_inference(run_name, sample_id)
    except ValueError as exc:
        return templates.TemplateResponse(
            request=request,
            name="test.html",
            context={
                "page_title": "Test",
                "runs": runs,
                "samples": samples,
                "selected_run_name": run_name,
                "error_message": str(exc),
            },
            status_code=400,
        )

    return templates.TemplateResponse(
        request=request,
        name="test.html",
        context={
            "page_title": "Test",
            "runs": runs,
            "samples": samples,
            "selected_run_name": run_name,
            "selected_sample_id": sample_id,
            "result": result,
        },
    )
```
